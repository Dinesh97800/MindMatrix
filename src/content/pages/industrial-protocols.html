use mind_matrix_core::protocols::{EtherCAT, Modbus};

// Initialize deterministic master stack
let mut master = EtherCATMaster::new("eth0")?;

// Map cyclic process data (PDO)
master.map_pdo(0x6040, 0x00, "ControlWord")?;
master.map_pdo(0x6041, 0x00, "StatusWord")?;

// Set cycle time to 250 microseconds
master.set_cycle_time(Duration::from_micros(250));

// Start isochronous real-time loop
master.start_rt_loop(|data| {
    let status = data.read_u16("StatusWord");
    if status & 0x08 != 0 {
        data.write_u16("ControlWord", 0x0F);
    }
});
